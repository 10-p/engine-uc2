#=============================================================================
# UC2GLDrv - the OpenGL render device: fixed-function stages, pixel-shader materials, post-processing.
#
# One source tree, compiled once. The GL binding is chosen at run time by the device class (Inc/UC2GLBinding.h):
# this package's device binds desktop OpenGL, its entry points looked up through SDL_GL_GetProcAddress into the
# device's pointer table; Source/UC2GL4ESDrv's subclass binds GL4ES on OpenGL ES / WebGL. No GL library is
# linked: SDL loads libGL at run time.
#=============================================================================

project(UC2GLDrv CXX)

file(GLOB SRC_FILES CONFIGURE_DEPENDS "Src/*.cpp")
add_library(${PROJECT_NAME} ${LIB_TYPE} ${SRC_FILES})

target_include_directories(${PROJECT_NAME}
  PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/Inc
  PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/Src
  # the GLSL translations of the Xbox pixel shaders, generated by tools/psh2glsl.py --emit
  PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/Shaders
)

# GPackage names the symbol IMPLEMENT_PACKAGE defines under __STATIC_LINK (see Source/Core/CMakeLists.txt).
target_compile_definitions(${PROJECT_NAME} PRIVATE GPackage=GUC2GLDrvPackage ThisPackage=UC2GLDrv)

target_link_libraries(${PROJECT_NAME} PUBLIC Engine PRIVATE SDL2::SDL2-static)
